<script>

let playerName="";

let currentQuestion=null;

let score=0;
let correct=0;
let wrong=0;
let questionNo=0;

let level=1;
let levelCorrect=0;

let answered=false;

let startTime=0;
let timerInterval=null;


/* START */

function startPlayer(){

let name=
document.getElementById("playerName")
.value.trim();

if(!name){
alert("पहले अपना नाम लिखें।");
return;
}

playerName=name;

document.getElementById("playerDisplay")
.innerText=playerName;

document.getElementById("namePage")
.classList.add("hidden");

document.getElementById("gamePage")
.classList.remove("hidden");

newGame();

}


/* NEW GAME */

function newGame(){

score=0;
correct=0;
wrong=0;
questionNo=0;

level=1;
levelCorrect=0;

startTime=Date.now();

clearInterval(timerInterval);

timerInterval=setInterval(updateTimer,1000);

updateTimer();

nextQuestion();

}


/* RANDOM */

function random(min,max){

return Math.floor(
Math.random()*(max-min+1)
)+min;

}


/* GENERATE QUESTION */

function generateQuestion(){

let type;
let a,b,c;
let answer;
let text;


/*
LEVEL 1
2 digit addition/subtraction
*/

if(level===1){

type=random(1,2);

if(type===1){

a=random(10,99);
b=random(10,99);

answer=a+b;

text=a+" + "+b;

}else{

a=random(20,99);
b=random(10,a);

answer=a-b;

text=a+" − "+b;

}

}


/*
LEVEL 2
2 digit multiplication
*/

else if(level===2){

type=random(1,3);

if(type===1){

a=random(10,99);
b=random(10,99);

answer=a+b;

text=a+" + "+b;

}

else if(type===2){

a=random(20,99);
b=random(10,a);

answer=a-b;

text=a+" − "+b;

}

else{

a=random(10,30);
b=random(2,9);

answer=a*b;

text=a+" × "+b;

}

}


/*
LEVEL 3
Bigger multiplication + division
*/

else if(level===3){

type=random(1,3);

if(type===1){

a=random(20,99);
b=random(10,30);

answer=a*b;

text=a+" × "+b;

}

else if(type===2){

b=random(2,12);
answer=random(10,50);

a=b*answer;

text=a+" ÷ "+b;

}

else{

a=random(100,999);
b=random(10,99);

answer=a+b;

text=a+" + "+b;

}

}


/*
LEVEL 4
Mixed
*/

else if(level===4){

type=random(1,4);

if(type===1){

a=random(100,999);
b=random(100,999);

answer=a+b;

text=a+" + "+b;

}

else if(type===2){

a=random(200,999);
b=random(50,a);

answer=a-b;

text=a+" − "+b;

}

else if(type===3){

a=random(10,50);
b=random(2,20);

answer=a*b;

text=a+" × "+b;

}

else{

b=random(2,20);
answer=random(10,100);

a=b*answer;

text=a+" ÷ "+b;

}

}


/*
LEVEL 5
PERCENTAGE + MIX
*/

else{

type=random(1,4);

if(type===1){

let p=random(5,50)*5;
let amount=random(2,20)*100;

answer=(p/100)*amount;

text=p+"% of "+amount;

}

else if(type===2){

a=random(100,999);
b=random(10,50);

answer=a+b;

text=a+" + "+b;

}

else if(type===3){

a=random(20,100);
b=random(2,20);

answer=a*b;

text=a+" × "+b;

}

else{

b=random(5,25);
answer=random(10,100);

a=b*answer;

text=a+" ÷ "+b;

}

}


return{
text:text,
answer:answer
};

}


/* NEXT QUESTION */

function nextQuestion(){

currentQuestion=generateQuestion();

questionNo++;

answered=false;

document.getElementById("questionText")
.innerText=currentQuestion.text;

document.getElementById("answer").value="";

document.getElementById("result").style.display="none";

document.getElementById("answer").focus();

updateDisplay();

}


/* ENTER */

function handleEnter(event){

if(event.key==="Enter"){

event.preventDefault();

checkAnswer();

}

}


/* CHECK */

function checkAnswer(){

if(answered)return;

if(!currentQuestion)return;

let input=
document.getElementById("answer")
.value.trim();

if(input==="")return;

let userAnswer=Number(input);

answered=true;

let result=
document.getElementById("result");

let title=
document.getElementById("resultTitle");

let resultText=
document.getElementById("resultText");


result.style.display="block";


/* CORRECT */

if(
Math.abs(
userAnswer-currentQuestion.answer
)<0.000001
){

correct++;
levelCorrect++;
score+=10;

result.className="result correct";

title.innerText="🎉 सही जवाब!";

resultText.innerHTML=
"बहुत बढ़िया! <b>"+
currentQuestion.answer+
"</b>";

/*
हर 20 सही पर level up
*/

if(levelCorrect>=20){

level++;

levelCorrect=0;

if(level>5){
level=5;
}

setTimeout(function(){

showLevelUp();

},500);

}

}


/* WRONG */

else{

wrong++;

score=Math.max(0,score-2);

result.className="result wrong";

title.innerText="❌ गलत जवाब";

resultText.innerHTML=
"सही जवाब था: <b>"+
currentQuestion.answer+
"</b>";

}


/*
1 second बाद नया question
सही/गलत दोनों में
*/

setTimeout(function(){

nextQuestion();

},1000);


updateDisplay();

}


/* LEVEL UP MESSAGE */

function showLevelUp(){

let names={
2:"Level 2 — Multiplication",
3:"Level 3 — Division",
4:"Level 4 — Mixed Maths",
5:"Level 5 — Percentage + Mixed"
};

alert(
"🎉 LEVEL UP!\n\n"+
"अब आप "+names[level]+" पर पहुँच गए!"
);

updateDisplay();

}


/* DISPLAY */

function updateDisplay(){

document.getElementById("score")
.innerText=score;

document.getElementById("correct")
.innerText=correct;

document.getElementById("wrong")
.innerText=wrong;

document.getElementById("questionNo")
.innerText=questionNo;

document.getElementById("levelName")
.innerText="Level "+level;

document.getElementById("levelProgress")
.innerText=levelCorrect;


/*
Progress 0-20
*/

document.getElementById("progressBar")
.style.width=
(levelCorrect/20*100)+"%";


let description="";

if(level===1){
description="2-digit Addition & Subtraction";
}

else if(level===2){
description="2-digit + Multiplication";
}

else if(level===3){
description="Multiplication + Division";
}

else if(level===4){
description="Mixed Maths";
}

else{
description="Percentage + Mixed Maths";
}

document.getElementById("levelDescription")
.innerText=description;

}


/* TIMER */

function updateTimer(){

if(!startTime)return;

let total=
Math.floor(
(Date.now()-startTime)/1000
);

let minutes=
Math.floor(total/60);

let seconds=
total%60;

document.getElementById("timer")
.innerText=
String(minutes).padStart(2,"0")+
":"+
String(seconds).padStart(2,"0");

}


/* FINISH */

function finishGame(){

if(questionNo===0)return;

clearInterval(timerInterval);

let totalSeconds=
Math.floor(
(Date.now()-startTime)/1000
);

let minutes=
Math.floor(totalSeconds/60);

let seconds=
totalSeconds%60;

let time=
String(minutes).padStart(2,"0")+
":"+
String(seconds).padStart(2,"0");


let records=
JSON.parse(
localStorage.getItem("lawMathRecords")||"[]"
);


records.unshift({

name:playerName,

time:time,

questions:questionNo,

correct:correct,

wrong:wrong,

level:level,

score:score,

date:new Date()
.toLocaleDateString("en-IN")

});


records=records.slice(0,50);


localStorage.setItem(
"lawMathRecords",
JSON.stringify(records)
);


displayRecords();


alert(
"🏆 Challenge Complete!\n\n"+
"Name: "+playerName+
"\nTime: "+time+
"\nQuestions: "+questionNo+
"\nCorrect: "+correct+
"\nWrong: "+wrong+
"\nLevel: "+level+
"\nScore: "+score
);


document.getElementById("gamePage")
.classList.add("hidden");

document.getElementById("namePage")
.classList.remove("hidden");

document.getElementById("playerName").value="";

playerName="";

}


/* RECORDS */

function displayRecords(){

let records=
JSON.parse(
localStorage.getItem("lawMathRecords")||"[]"
);

let tbody=
document.getElementById("records");

tbody.innerHTML="";


if(records.length===0){

tbody.innerHTML=
"<tr>"+
"<td colspan='7'>"+
"अभी कोई record नहीं है।"+
"</td>"+
"</tr>";

return;

}


records.forEach(function(r){

let tr=
document.createElement("tr");

tr.innerHTML=

"<td>"+escapeHTML(r.name)+"</td>"+

"<td>"+escapeHTML(r.time)+"</td>"+

"<td>"+r.questions+"</td>"+

"<td>✅ "+r.correct+"</td>"+

"<td>❌ "+r.wrong+"</td>"+

"<td>Level "+r.level+"</td>"+

"<td><b>"+r.score+"</b></td>";

tbody.appendChild(tr);

});

}


/* SECURITY */

function escapeHTML(value){

return String(value)

.replaceAll("&","&amp;")

.replaceAll("<","&lt;")

.replaceAll(">","&gt;")

.replaceAll('"',"&quot;")

.replaceAll("'","&#039;");

}


displayRecords();
